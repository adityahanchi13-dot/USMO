local function ry(d) return CFrame.Angles(0, math.rad(d), 0) end

local function createBodyWeld(part0, part1)
    local weld = Instance.new("WeldConstraint")
    weld.Part0 = part0
    weld.Part1 = part1
    weld.Parent = part0
    return weld
end
-- the part maker: exactly the steps the working car does for each of its parts
local function newKit(car, root)
    local k = { parts = {}, wheels = {}, rampParts = {} }
    function k.at(x, y, z) return root * CFrame.new(x * S, y * S, z * S) end
    function k.off(x, y, z) return CFrame.new(x * S, y * S, z * S) end
    function k.facing(x, y, z, dx, dy, dz)
        local p = Vector3.new(x, y, z) * S
        return root * CFrame.new(p, p + Vector3.new(dx, dy, dz))
    end
    function k.make(class, shape, name, size, cf, color, mat, weldTo)
        local p = Instance.new(class)
        p.Name = name
        if shape then p.Shape = shape end
        p.Size = size * S
        p.CFrame = cf
        p.Color = color
        p.Material = mat
        p.Anchored = true
        p.CanCollide = false
        p.Parent = car
        f(p)
        if weldTo ~= false then createBodyWeld(weldTo or k.chassis, p) end
        table.insert(k.parts, p)
        return p
    end
    function k.block(name, sx, sy, sz, cf, color, mat, weldTo)
        return k.make("Part", nil, name, Vector3.new(sx, sy, sz), cf, color, mat, weldTo)
    end
    function k.wedge(name, sx, sy, sz, cf, color, mat, weldTo)
        return k.make("WedgePart", nil, name, Vector3.new(sx, sy, sz), cf, color, mat, weldTo)
    end
    function k.cyl(name, len, d, cf, color, mat, weldTo)
        return k.make("Part", CYL, name, Vector3.new(len, d, d), cf, color, mat, weldTo)
    end
    return k
end

-- bolted armour tiles on the sides, bow cheeks and upper glacis
local function buildArmour(k)
    local at, block = k.at, k.block
    local slope = math.deg(math.atan(1.3 / 4.5))
    for _, s in ipairs({ -1, 1 }) do
        for col = 0, 6 do
            for row = 0, 1 do
                -- on the right, the exhaust grille takes the place of the first two lower tiles
                if not (s > 0 and row == 0 and col <= 1) then
                    block("ArmourTile", 0.12, 1.3, 2.55, at(s * 4.36, 4.6 + row * 1.37, -5.675 + col * 2.62), C.TAN, M.SMOOTH)
                end
            end
        end
        block("BowTile", 0.12, 1.4, 3.6, at(s * 4.36, 4.5, -9.6), C.TAN, M.SMOOTH)
    end
    for ix = -1, 1 do
        for iz = 0, 1 do
            local z = D.NOSE + 1.2 + iz * 2.1
            local y = 5.4 + 1.3 * (z - D.NOSE) / 4.5 + 0.06
            block("GlacisTile", 2.6, 0.12, 2.0, at(ix * 2.75, y, z) * rx(-slope), C.TAN, M.SMOOTH)
        end
    end
end

-- 8 wheels on Welds from the chassis (like the car's), so the drive loop can steer and roll them
local function buildWheels(k)
    local at, cyl = k.at, k.cyl
    local r = D.WHEEL_R
    local iceFriction = PhysicalProperties.new(0, 0, 0, 0, 0)
    for i, z in ipairs(D.AXLES) do
        local steer = 0
        if i == 1 then steer = 1 elseif i == 2 then steer = 0.55 end
        for _, s in ipairs({ -1, 1 }) do
            local x = s * D.WHEEL_X
            local wheel = cyl("StrykerWheel", 1.3, r * 2, at(x, r, z), C.RUBBER, M.RUBBER, false)
            wheel.CanCollide = true
            wheel.CustomPhysicalProperties = iceFriction
            local face = x + s * 0.65
            cyl("Rim", 0.1, r * 1.2, at(face, r, z), C.TAN, M.SMOOTH, wheel)
            cyl("Hub", 0.28, r * 0.55, at(face + s * 0.12, r, z), C.TAN_DARK, M.METAL, wheel)
            cyl("HubCap", 0.12, r * 0.25, at(face + s * 0.3, r, z), C.STEEL, M.METAL, wheel)
            local weld = Instance.new("Weld")
            weld.Part0 = k.chassis
            weld.Part1 = wheel
            weld.C0 = CFrame.new(x * S, (r - D.CH_Y) * S, (z - D.CH_Z) * S)
            weld.C1 = CFrame.new()
            weld.Parent = k.chassis
            table.insert(k.wheels, { weld = weld, base = weld.C0, steer = steer })
        end
    end
end
-- the rear ramp: a Weld hinged at the floor; the drive loop swings it by changing C0 (like the car steers)
local function buildRamp(k)
    local w, h = D.RAMP_W, D.RAMP_H
    local hingeY, hingeZ = 2.0, D.TAIL - 0.15
    local cf = k.at(0, hingeY + h / 2, hingeZ)
    local off, block = k.off, k.block
    local ramp = k.make("Part", nil, "Ramp", Vector3.new(w, h, 0.3), cf, C.TAN, M.SMOOTH, false)
    ramp.CanCollide = true -- troops walk up it
    local liner = block("RampLiner", w - 0.3, h - 0.3, 0.05, cf * off(0, 0, -0.17), C.INTERIOR, M.SMOOTH, ramp)
    local door = block("RampDoor", 1.8, 3.1, 0.05, cf * off(0.6, -0.05, 0.17), C.TAN, M.SMOOTH, ramp)
    block("RampVisionBlock", 0.4, 0.25, 0.06, cf * off(0.6, 1.1, 0.2), C.GLASS, M.SMOOTH, ramp)
    block("RampHandle", 0.1, 0.5, 0.1, cf * off(1.25, 0.2, 0.24), C.BLACK, M.METAL, ramp)
    block("RampHinge", w - 0.4, 0.25, 0.25, cf * off(0, -h / 2 + 0.12, 0.18), C.STEEL, M.METAL, ramp)
    for i = 0, 3 do block("RampTread", w - 0.8, 0.08, 0.06, cf * off(0, -1.5 + i * 0.9, -0.21), C.STEEL, M.METAL, ramp) end
    local hinge = CFrame.new(0, (hingeY - D.CH_Y) * S, (hingeZ - D.CH_Z) * S)
    local weld = Instance.new("Weld")
    weld.Part0 = k.chassis
    weld.Part1 = ramp
    weld.C0 = hinge
    weld.C1 = CFrame.new(0, -h / 2 * S, 0)
    weld.Parent = k.chassis
    k.rampWeld = weld
    k.rampBase = hinge
    k.rampParts = { ramp, liner, door }
end
local function destroyStryker(player)
    local entry = Strykers[player]
    if entry then
        for _, p in ipairs(entry.parts) do pcall(function() p:Destroy() end) end -- in case f() moved any out of the model
        pcall(function() entry.car:Destroy() end)
        Strykers[player] = nil
    end
end
-- the drive loop: the car's loop, plus all 8 wheels rolling and the ramp swinging
local function drive(entry)
    local car, chassis, seat = entry.car, entry.chassis, entry.driverSeat
    local speed, steerAngle, rolled, ramp = 0, 0, 0, 0
    local shownSteer, shownRolled = 0, 0
    local last = tick()
    local told = false
    local function step(dt)
        local velocity = chassis.AssemblyLinearVelocity
        if seat.Occupant then
            local throttle = seat.Throttle
            local steer = seat.Steer
            local target = throttle * DRIVE.MAX_SPEED
            if throttle < 0 then target = throttle * DRIVE.REVERSE end
            speed = speed + (target - speed) * DRIVE.ACCEL
            steerAngle = steerAngle + (-steer * DRIVE.MAX_STEER - steerAngle) * 0.15
            if steer == 0 and math.abs(steerAngle) < 0.1 then steerAngle = 0 end
            chassis.AssemblyLinearVelocity = chassis.CFrame.LookVector * speed + Vector3.new(0, velocity.Y, 0)
            local ratio = math.clamp(math.abs(speed) / DRIVE.MAX_SPEED, 0.05, 1)
            if speed < -0.5 then ratio = -ratio end -- reversing turns the other way, like a real vehicle
            chassis.AssemblyAngularVelocity = Vector3.new(0, -steer * DRIVE.TURN * ratio, 0)
        else
            -- nobody driving: roll to a stop and stay put (the tyres have no friction)
            speed = speed * 0.8
            if math.abs(speed) < 0.05 then speed = 0 end
            steerAngle = 0
            local spin = chassis.AssemblyAngularVelocity
            chassis.AssemblyLinearVelocity = chassis.CFrame.LookVector * speed + Vector3.new(0, velocity.Y, 0)
            chassis.AssemblyAngularVelocity = Vector3.new(spin.X, 0, spin.Z)
        end
        rolled = rolled + speed * dt
        if math.abs(rolled - shownRolled) > 0.05 or steerAngle ~= shownSteer then
            shownRolled, shownSteer = rolled, steerAngle
            local roll = CFrame.Angles(-rolled / (D.WHEEL_R * S), 0, 0)
            for _, w in ipairs(entry.wheels) do
                w.weld.C0 = w.base * CFrame.Angles(0, math.rad(steerAngle) * w.steer, 0) * roll
            end
        end
        if math.abs(speed) > 3 then entry.rampOpen = false end
        local goal = 0
        if entry.rampOpen then goal = D.RAMP_OPEN end
        if ramp ~= goal then
            local move = D.RAMP_OPEN * dt / DRIVE.RAMP_TIME
            if ramp < goal then ramp = math.min(goal, ramp + move) else ramp = math.max(goal, ramp - move) end
            entry.rampWeld.C0 = entry.rampBase * CFrame.Angles(ramp, 0, 0)
        end
    end
    while car and car.Parent do
        task.wait(0.03)
        local now = tick()
        local dt = math.min(now - last, 0.2)
        last = now
        local ok, err = pcall(step, dt)
        if not ok and not told then
            told = true
            print("[Stryker] drive error: " .. tostring(err))
        end
    end
end
local function toggleRamp(entry)
    entry.rampOpen = not entry.rampOpen
end
local function spawnStryker(player)
    destroyStryker(player)
    local playerCF = getPlayerPosition(player)
    if not playerCF then return end
    if typeof(playerCF) == "Vector3" then playerCF = CFrame.new(playerCF) end
    local root = playerCF * CFrame.new(0, 0, -26) -- ground level of the Stryker, in front of the player
    local car = Instance.new("Model")
    car.Name = player .. "_UnarmedStryker"
    f(car)
    local k = newKit(car, root)
    local ok, err = pcall(function()
        buildHull(k)
        buildArmour(k)
        buildBow(k)
        buildRoof(k)
        buildGun(k)
        buildSidesAndRear(k)
        buildInterior(k)
        buildWheels(k)
        buildRamp(k)
    end)
    if not ok then
        print("[Stryker] build failed after " .. #k.parts .. " parts: " .. tostring(err))
        pcall(announce, "Stryker failed to build: " .. tostring(err), player)
        pcall(function() car:Destroy() end)
        return
    end
    -- unanchor the entire assembly
    for _, p in ipairs(k.parts) do p.Anchored = false end
    local entry = {
        car = car, parts = k.parts, chassis = k.chassis, driverSeat = k.driverSeat, commanderSeat = k.commanderSeat,
        troopSeats = k.troopSeats, wheels = k.wheels, rampWeld = k.rampWeld, rampBase = k.rampBase, rampOpen = false,
    }
    Strykers[player] = entry
    for _, part in ipairs(k.rampParts) do
        pcall(function()
            local detector = Instance.new("ClickDetector")
            detector.MaxActivationDistance = 30
            detector.Parent = part
            detector.MouseClick:Connect(function() toggleRamp(entry) end)
        end)
    end
    task.spawn(drive, entry)
    print("[Stryker] built " .. #k.parts .. " parts for " .. player)
    pcall(setPlayerPosition, player, k.driverSeat.CFrame * CFrame.new(0, 4, 0))
    pcall(announce, player .. " deployed an Unarmed Stryker. W/S drive, A/D steer. Click the ramp or type :ramp.", player)
end
-- the Stryker nearest to a player (within 60 studs)
local function nearest(player)
    local cf = getPlayerPosition(player)
    if not cf then return nil end
    local pos = cf.Position
    if typeof(cf) == "Vector3" then pos = cf end
    local best, bestDist = nil, 60
    for _, entry in pairs(Strykers) do
        local ok, d = pcall(function() return (entry.chassis.Position - pos).Magnitude end)
        if ok and d < bestDist then best, bestDist = entry, d end
    end
    return best
end
-- drop the player onto a free seat (they fall through the roof onto it, the way the car seats its owner)
local function board(player, role)
    local entry = nearest(player)
    if not entry then
        pcall(announce, "No Stryker close enough.", player)
        return
    end
    local seats = entry.troopSeats
    if role == "driver" then seats = { entry.driverSeat } elseif role == "commander" then seats = { entry.commanderSeat } end
    for _, seat in ipairs(seats) do
        if not seat.Occupant then
            pcall(setPlayerPosition, player, seat.CFrame * CFrame.new(0, 3, 0))
            return
        end
    end
    pcall(announce, "That seat is taken.", player)
end
event("chatted", function(data)
    local player = data.Value[1]
    local message = data.Value[2]
    if not player or not message then return end
    local command = string.match(string.lower(message), "^%s*(.-)%s*$")
    if command == ":spawn stryker" then
        spawnStryker(player)
    elseif command == ":despawn stryker" then
        destroyStryker(player)
    elseif command == ":ramp" then
        local entry = nearest(player)
        if entry then toggleRamp(entry) end
    else
        local role = string.match(command, "^:stryker (%a+)$")
        if role == "driver" or role == "commander" or role == "board" then board(player, role) end
    end
end)
print("[Stryker] Unarmed Stryker loaded. Type :spawn stryker")
