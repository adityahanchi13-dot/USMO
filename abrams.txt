-- ===== MATH / CFrame HELPERS =====
local function rx(deg) return CFrame.Angles(math.rad(deg), 0, 0) end
local function ry(deg) return CFrame.Angles(0, math.rad(deg), 0) end
local function rz(deg) return CFrame.Angles(0, 0, math.rad(deg)) end
local function atan2(y, x)
  if math.atan2 then return math.atan2(y, x) end
  return math.atan(y, x)
end

-- ===== WELD HELPERS =====
-- WeldConstraint: rigid follow. Used for all detail parts (they don't animate).
local function weldConstraint(part0, part1)
  local w = Instance.new("WeldConstraint")
  w.Part0 = part0
  w.Part1 = part1
  w.Parent = part0
  return w
end
-- Weld: animatable. Used ONLY for turretPivot and gunPivot.
local function weld(part0, part1, c0)
  local w = Instance.new("Weld")
  w.Part0 = part0
  w.Part1 = part1
  w.C0 = c0 or CFrame.new()
  w.C1 = CFrame.new()
  w.Parent = part0
  return w
end

-- ===== BUILD KIT =====
local function newKit(model, root)
  local k = { parts = {}, dust = {}, spinners = {} }
  function k.at(x, y, z) return root * CFrame.new(x*SCALE, y*SCALE, z*SCALE) end
  function k.off(x, y, z) return CFrame.new(x*SCALE, y*SCALE, z*SCALE) end
  function k.facing(x, y, z, dx, dy, dz)
    local p = Vector3.new(x, y, z) * SCALE
    return root * CFrame.new(p, p + Vector3.new(dx, dy, dz))
  end
  function k.make(class, name, size, cf, color, material, weldTo, shape)
    local p = Instance.new(class)
    p.Name = name
    if shape then p.Shape = shape end
    p.Size = size * SCALE
    p.CFrame = cf
    p.Color = color
    p.Material = material
    p.Anchored = true
    p.CanCollide = false
    p.CanTouch = false
    p.Massless = true
    p.Parent = model
    f(p)
    if weldTo then weldConstraint(weldTo, p) end
    table.insert(k.parts, p)
    return p
  end
  function k.block(name, sx, sy, sz, cf, color, material, weldTo)
    return k.make("Part", name, Vector3.new(sx, sy, sz), cf, color, material, weldTo, false)
  end
  function k.wedge(name, sx, sy, sz, cf, color, material, weldTo)
    return k.make("WedgePart", name, Vector3.new(sx, sy, sz), cf, color, material, weldTo, false)
  end
  function k.cylinder(name, length, diameter, cf, color, material, weldTo)
    return k.make("Part", name, Vector3.new(length, diameter, diameter), cf, color, material, weldTo, CYL)
  end
  function k.bolt(cf, weldTo)
    return k.cylinder("Bolt", 0.06, 0.14, cf, COLOR.TAN_DARK, MAT.METAL, weldTo)
  end
  -- (new) a road wheel, idler or sprocket that spins with its track: held to the chassis by a Weld
  -- (instead of a WeldConstraint) so the drive loop can turn it. `part` must be built unrotated.
  function k.axle(part, x, y, z, side, radius)
    local j = Instance.new("Weld")
    j.Name = "Axle"
    j.Part0 = k.chassis
    j.Part1 = part
    j.C0 = k.off(x, y - DIM.CHASSIS_Y, z - DIM.CHASSIS_Z)
    j.C1 = CFrame.new()
    j.Parent = k.chassis
    table.insert(k.spinners, { weld = j, base = j.C0, side = side, radius = radius })
  end
  function k.seat(class, name, cf, weldTo)
    local s = k.make(class, name, Vector3.new(1.5, 0.4, 1.4), cf, COLOR.BLACK, MAT.FABRIC, weldTo, false)
    s.CanCollide = true
    s.CanTouch = true
    s.Transparency = 1
    return s
  end
  return k
end

-- ===== SEAT PROMPT =====
-- (fix) The seats sit inside the hull, on top of the 3.75-stud-tall chassis, so walking into them is
-- almost impossible. Each seat gets an "[E] Sit" prompt, like the Humvee and Stryker.
local function addSeatPrompt(seat, label)
  pcall(function()
    local prompt = Instance.new("ProximityPrompt")
    prompt.Name = "SitPrompt"
    prompt.ActionText = "Sit"
    prompt.ObjectText = label
    prompt.KeyboardKeyCode = Enum.KeyCode.E
    prompt.HoldDuration = 0
    prompt.MaxActivationDistance = 12
    prompt.RequiresLineOfSight = false
    prompt.Parent = seat
    prompt.Triggered:Connect(function(who)
      if seat.Occupant then return end
      pcall(function()
        local char = who.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum then seat:Sit(hum) end
      end)
    end)
  end)
end

-- ===== RUNNING GEAR =====
local function trackRun(k, x, y1, z1, y2, z2)
  local dy, dz = y2 - y1, z2 - z1
  local length = math.sqrt(dy * dy + dz * dz)
  local count = math.max(1, math.floor(length / DIM.SHOE_PITCH + 0.5))
  local tilt = math.deg(atan2(-dy, dz))
  for i = 0, count - 1 do
    local t = (i + 0.5) / count
    k.block("TrackShoe", DIM.TRACK_W, 0.25, length / count - DIM.SHOE_GAP,
      k.at(x, y1 + dy * t, z1 + dz * t) * rx(tilt), COLOR.TRACK, MAT.METAL, k.chassis)
  end
end

local function trackArc(k, x, cy, cz, radius, fromDeg, toDeg)
  local sweep = toDeg - fromDeg
  local arc = radius * math.rad(math.abs(sweep))
  local count = math.max(1, math.floor(arc / DIM.SHOE_PITCH + 0.5))
  for i = 0, count - 1 do
    local a = fromDeg + sweep * (i + 0.5) / count
    local r = math.rad(a)
    k.block("TrackShoe", DIM.TRACK_W, 0.25, arc / count - DIM.SHOE_GAP,
      k.at(x, cy + radius * math.cos(r), cz + radius * math.sin(r)) * rx(a), COLOR.TRACK, MAT.METAL, k.chassis)
  end
end

local function lowerTangent(az, ay, ar, bz, by, br)
  local dz, dy = bz - az, by - ay
  local base = atan2(dy, dz)
  local spread = math.acos((ar - br) / math.sqrt(dz * dz + dy * dy))
  local phi = base + spread
  if math.sin(base - spread) < math.sin(phi) then phi = base - spread end
  local nz, ny = math.cos(phi), math.sin(phi)
  return az + ar * nz, ay + ar * ny, bz + br * nz, by + br * ny, math.deg(atan2(nz, ny))
end

  local outX = s * (DIM.TRACK_X + DIM.DUAL_OFFSET)

  local iceFriction = PhysicalProperties.new(0, 0, 0, 0, 0)
  local wy = DIM.WHEEL_Y
  for _, wz in ipairs(DIM.WHEEL_Z) do
    local wheel = cylinder("RoadWheel", 0.75, DIM.WHEEL_R * 2, at(outX, wy, wz), COLOR.RUBBER, MAT.RUBBER, false)
    wheel.CanCollide = true
    wheel.CustomPhysicalProperties = iceFriction

    cylinder("RoadWheelInner", 0.75, DIM.WHEEL_R * 2, at(inX, wy, wz), COLOR.RUBBER, MAT.RUBBER, wheel)

  end
  cylinder("IdlerHub", 0.14, 0.5, at(faceX + s * 0.1, iy, iz), COLOR.STEEL, MAT.METAL, idler)

      COLOR.TAN_DARK, MAT.METAL, sprocket)
  end

  local ir = DIM.IDLER_R + 0.13
  local sr = DIM.SPROCKET_R + 0.13
  local pathR = wy - 0.125
  local frontZ = DIM.WHEEL_Z[1]
  local backZ = DIM.WHEEL_Z[#DIM.WHEEL_Z]
  local fz1, fy1, fz2, fy2, frontWrap = lowerTangent(frontZ, wy, pathR, iz, iy, ir)
  local bz1, by1, bz2, by2, backWrap = lowerTangent(backZ, wy, pathR, sz, sy, sr)
  trackRun(k, tx, 0.125, fz1, 0.125, bz1)
  trackRun(k, tx, fy1, fz1, fy2, fz2)
  trackRun(k, tx, by1, bz1, by2, bz2)
  trackArc(k, tx, iy, iz, ir, 0, frontWrap)
  trackArc(k, tx, sy, sz, sr, 0, backWrap)
  local a = Vector3.new(tx, iy + ir, iz) * SCALE
  local b = Vector3.new(tx, sy + sr, sz) * SCALE
  k.make("Part", "TrackTop", Vector3.new(DIM.TRACK_W, 0.25, (b - a).Magnitude / SCALE),
    k.at(0, 0, 0) * CFrame.new((a + b) / 2, b), COLOR.TRACK, MAT.METAL, c, false)
end

-- ===== SKIRTS =====
local function buildSkirt(k, s)
  local c = k.chassis
  local at, block, wedge, bolt = k.at, k.block, k.wedge, k.bolt
  local kx = s * DIM.SKIRT_X
  local fx = s * 6.12
  local top, bottom = DIM.SKIRT_TOP, DIM.SKIRT_BOTTOM
  local h = top - bottom
  local midY = (top + bottom) / 2
  local edges = { -12.5, -8.95, -5.1, -1.9, 1.3, 4.9, 8.9, 12.8 }
  block("Skirt", 0.3, h, 20.5, at(kx, midY, -0.85), COLOR.TAN, MAT.SMOOTH, c)
  wedge("SkirtFrontRake", 0.3, h, 1.4, at(kx, midY, -11.8) * rz(180), COLOR.TAN, MAT.SMOOTH, c)
  block("SkirtRear", 0.3, top - 3.3, 3.4, at(kx, (top + 3.3) / 2, 11.1), COLOR.TAN, MAT.SMOOTH, c)
  wedge("SkirtRearRake", 0.3, 3.3 - bottom, 3.4, at(kx, (3.3 + bottom) / 2, 11.1) * rx(180), COLOR.TAN, MAT.SMOOTH, c)

  block("SkirtDust", 0.04, 0.3, 20.5, at(fx, bottom + 0.15, -0.85), COLOR.DUST, MAT.SMOOTH, c)
  for i = 2, #edges - 1 do
    block("SkirtSeam", 0.04, h, 0.08, at(fx, midY, edges[i]), COLOR.BLACK, MAT.SMOOTH, c)
  end
  for i = 3, #edges - 1 do
    for _, t in ipairs({0.25, 0.5, 0.75}) do
      bolt(at(s * 6.13, top - 0.3, edges[i] + t * (edges[i + 1] - edges[i])), c)
    end
  end
  local armourZ = (-11.1 + edges[3]) / 2
  local armourLen = edges[3] + 11.1
  local ah = h - 0.2
  block("SkirtArmour", 0.18, ah, armourLen, at(s * 6.19, midY, armourZ), COLOR.TAN, MAT.SMOOTH, c)
  wedge("SkirtArmourRake", 0.18, ah, 1.26, at(s * 6.19, midY, -11.73) * rz(180), COLOR.TAN, MAT.SMOOTH, c)

  for i = 1, 2 do
    local z0 = math.max(edges[i], -11.1)
    for _, t in ipairs({0.2, 0.5, 0.8}) do
      local z = z0 + t * (edges[i + 1] - z0)
      bolt(at(s * 6.31, top - 0.3, z), c)
      bolt(at(s * 6.31, bottom + 0.45, z), c)
    end
  end
end

-- ===== DUST =====
local function buildDust(k)
  for _, s in ipairs({-1, 1}) do
    local emitter = k.block("DustEmitter", 0.3, 0.3, 0.3, k.at(s * DIM.TRACK_X, 0.6, 10.6), COLOR.DUST, MAT.SMOOTH, k.chassis)
    emitter.Transparency = 1
    local ok, smoke = pcall(function()
      local sm = Instance.new("Smoke")
      sm.Color = COLOR.DUST
      sm.Opacity = 0
      sm.RiseVelocity = 2
      sm.Size = 3
      sm.Enabled = false
      sm.Parent = emitter
      return sm
    end)
    if ok then table.insert(k.dust, smoke) end
  end
end

-- ===== ASSEMBLE THE TANK =====
local function buildTank(owner, root, entry)
  local car = Instance.new("Model")
  car.Name = owner .. "_Abrams"
  f(car)
  entry.car = car -- (fix) known straight away, so a failed build is cleaned up instead of left in the map

  local k = newKit(car, root)

  -- Chassis (main collider, has mass)
  local chassis = k.block("Chassis", 7.2, 1.4, 22.4, k.at(0, DIM.CHASSIS_Y, DIM.CHASSIS_Z), COLOR.TAN, MAT.SMOOTH, false)
  chassis.Massless = false
  chassis.CanCollide = true
  k.chassis = chassis

  -- Detailed hull, running gear, skirts
  buildHull(k)
  for _, s in ipairs({-1, 1}) do
    buildRunningGear(k, s)
    buildSkirt(k, s)
  end

  -- Turret pivot (invisible, animated via Weld)
  local turretPivot = k.block("TurretPivot", 0.4, 0.2, 0.4, k.at(0, DIM.PIVOT_Y, DIM.RING_Z), COLOR.BLACK, MAT.SMOOTH, false)
  turretPivot.Transparency = 1
  local turretWeld = weld(chassis, turretPivot,
    k.off(0, DIM.PIVOT_Y - DIM.CHASSIS_Y, DIM.RING_Z - DIM.CHASSIS_Z))

  -- Gun pivot (invisible, animated via Weld, child of turretPivot)
  local gunPivot = k.block("GunPivot", 0.3, 0.3, 0.3, k.at(0, DIM.GUN_Y, DIM.GUN_PIVOT_Z), COLOR.BLACK, MAT.SMOOTH, false)
  gunPivot.Transparency = 1
  local gunWeld = weld(turretPivot, gunPivot,
    k.off(0, DIM.GUN_Y - DIM.PIVOT_Y, DIM.GUN_PIVOT_Z - DIM.RING_Z))

  k.turretPivot = turretPivot
  k.gunPivot = gunPivot

  -- Turret ring (visual, welded to chassis)
  k.cylinder("TurretRing", 0.4, 7.0,
    k.at(0, (DIM.DECK_Y + DIM.TURRET_Y) / 2, DIM.RING_Z) * rz(90),
    COLOR.BLACK, MAT.METAL, chassis)

  -- Detailed turret, stowage, gun
  buildTurret(k)
  buildTurretStowage(k)
  buildGun(k)
  buildDust(k)

  -- Seats: driver on the left, gunner on the right, both welded to chassis
  local driverSeat = k.seat("VehicleSeat", "DriverSeat", k.at(-1.3, DIM.SEAT_Y, DIM.RING_Z), chassis)
  local gunnerSeat = k.seat("VehicleSeat", "GunnerSeat", k.at(1.3, DIM.SEAT_Y, DIM.RING_Z), chassis)
  for _, s in ipairs({driverSeat, gunnerSeat}) do
    s.MaxSpeed = 0
    s.Torque = 0
    s.TurnSpeed = 0
    s.HeadsUpDisplay = false
  end

  -- Unanchor everything. The chassis is the physical driver; everything else
  -- is welded to it and follows.
  for _, p in ipairs(k.parts) do
    p.Anchored = false
  end
  turretPivot.Anchored = false
  gunPivot.Anchored = false

  addSeatPrompt(driverSeat, "Abrams driver")
  addSeatPrompt(gunnerSeat, "Abrams gunner")

  -- (fix) Roblox puts a tank that sits still "to sleep", and a sleeping tank can ignore the velocity the
  -- drive loop sets, so it feels locked after you get out and back in. When someone gets in, it is
  -- lifted 0.05 studs, which wakes it. Only a CFrame nudge, no push, at most once every 2 s.
  local lastWake = 0
  local function wake()
    if tick() - lastWake < 2 then return end
    lastWake = tick()
    pcall(function() chassis.CFrame = chassis.CFrame + Vector3.new(0, 0.05, 0) end)
  end

  -- ===== DRIVE LOOP (Humvee pattern) =====
  -- (new, from the old detailed tank) each track runs at its own speed, faster on the outside of a turn and
  -- backwards when pivoting; its road wheels, idler and sprocket spin to match, and dust kicks up behind.
  local spinners = k.spinners
  local dust = k.dust
  local halfTrack = DIM.TRACK_X * SCALE
  task.spawn(function()
    local speed = 0
    local turn = 0
    local wasManned = false
    local last = tick()
    local rolled = { [-1] = 0, [1] = 0 } -- distance each track has run, studs
    local lastSpin, lastDust = 0, 0
    local dustOn = false
    while car.Parent do
      local driving = driverSeat.Occupant ~= nil
      -- (fix) full speed only while driven or still moving; four checks a second when parked
      if driving or speed ~= 0 or turn ~= 0 then task.wait(0.03) else task.wait(0.25) end
      local now = tick()
      local dt = math.min(now - last, 0.3)
      last = now

      local manned = driving or gunnerSeat.Occupant ~= nil
      if manned and not wasManned then wake() end
      wasManned = manned

      if driving then
        local throttle = driverSeat.Throttle
        local steer    = driverSeat.Steer
        local targetSpeed = throttle * MAX_SPEED
        if throttle < 0 then targetSpeed = throttle * REVERSE_SPEED end
        speed = speed + (targetSpeed - speed) * ACCELERATION
        if throttle == 0 and math.abs(speed) < 0.05 then speed = 0 end
        turn  = turn  + (-steer * TURN_RATE - turn) * 0.15
        if steer == 0 and math.abs(turn) < 0.002 then turn = 0 end
      else
        speed = speed * 0.9
        if math.abs(speed) < 0.05 then speed = 0 end
        turn = 0
      end
      pcall(function()
        chassis.AssemblyLinearVelocity =
          chassis.CFrame.LookVector * speed + Vector3.new(0, chassis.AssemblyLinearVelocity.Y, 0)
        chassis.AssemblyAngularVelocity = Vector3.new(0, turn, 0)
      end)

      local left = speed - turn * halfTrack
      local right = speed + turn * halfTrack
      rolled[-1] = rolled[-1] + left * dt
      rolled[1] = rolled[1] + right * dt
      -- wheels redraw 10 times a second, and only while a track is moving
      if now - lastSpin >= 0.1 and (math.abs(left) > 0.05 or math.abs(right) > 0.05) then
        lastSpin = now
        pcall(function()
          for _, sp in ipairs(spinners) do
            sp.weld.C0 = sp.base * CFrame.Angles(-rolled[sp.side] / (sp.radius * SCALE), 0, 0)
          end
        end)
      end
      -- dust: only touched while it's kicking up, or the moment it stops
      if now - lastDust >= 0.3 then
        lastDust = now
        local churn = math.min(1, math.max(math.abs(left), math.abs(right)) / MAX_SPEED)
        local on = churn > 0.12
        if on or dustOn then
          for _, smoke in ipairs(dust) do
            pcall(function()
              smoke.Enabled = on
              smoke.Opacity = 0.05 + 0.2 * churn
              smoke.Size = 2 + 4 * churn
            end)
          end
        end
        dustOn = on
      end
    end
  end)

  -- ===== TURRET LOOP (aim only) =====
  task.spawn(function()
    local yaw   = 0
    local pitch = 0
    local yawBase   = turretWeld.C0
    local pitchBase = gunWeld.C0
    local last = tick()
    while car.Parent do
      -- (fix) four checks a second while nobody is on the gun
      if gunnerSeat.Occupant then task.wait(0.03) else task.wait(0.25) end
      local now = tick()
      local dt = math.min(now - last, 0.1) -- (fix) real time step, so aiming speed doesn't change with lag
      last = now
      if gunnerSeat.Occupant then
        local throttle = gunnerSeat.Throttle
        local steer    = gunnerSeat.Steer
        if steer ~= 0 or throttle ~= 0 then
          yaw   = (yaw - steer * TRAVERSE_SPEED * dt + math.pi) % (2 * math.pi) - math.pi
          pitch = math.clamp(pitch + throttle * ELEVATE_SPEED * dt, MIN_PITCH, MAX_PITCH)
          pcall(function()
            turretWeld.C0 = yawBase   * CFrame.Angles(0, yaw, 0)
            gunWeld.C0    = pitchBase * CFrame.Angles(pitch, 0, 0)
          end)
        end
      end
    end
  end)

  entry.chassis    = chassis
  entry.driverSeat = driverSeat
  entry.gunnerSeat = gunnerSeat
  return true
end

-- ===== SPAWN / DESTROY =====
local function destroyTank(player)
  if Tanks[player] then
    pcall(function() Tanks[player].car:Destroy() end)
    Tanks[player] = nil
  end
end

local function spawnTank(player)
  destroyTank(player)
  local cf = getPlayerPosition(player)
  if not cf then return end
  if typeof(cf) == "Vector3" then cf = CFrame.new(cf) end

  local feetY = cf.Position.Y - 3
  local forward = Vector3.new(cf.LookVector.X, 0, cf.LookVector.Z)
  if forward.Magnitude < 0.1 then forward = Vector3.new(0, 0, -1) end
  forward = forward.Unit

  local tankPos = Vector3.new(cf.Position.X, feetY, cf.Position.Z) + forward * 30
  local root = CFrame.new(tankPos + Vector3.new(0, 3, 0), tankPos + Vector3.new(0, 3, 0) + forward)

  local entry = {}
  Tanks[player] = entry

  local ok, err = pcall(buildTank, player, root, entry)
  if not ok then
    print("[Abrams] build failed: " .. tostring(err))
    pcall(function() entry.car:Destroy() end)
    Tanks[player] = nil
    return
  end

  -- Drop the player just above the driver seat
  local drop = root * CFrame.new(-1.3 * SCALE, DIM.SEAT_Y * SCALE + 3, DIM.RING_Z * SCALE)
  pcall(setPlayerPosition, player, drop)
  pcall(announce, player .. " deployed an Abrams. Driver: W/S + A/D. Right seat controls the turret. Press E near the tank to get in.", player)
end

-- ===== COMMANDS =====
event("chatted", function(data)
  local player = data.Value[1]
  local message = data.Value[2]
  if not player or not message then return end
  local cmd = string.match(string.lower(message), "^%s*(.-)%s*$")

  if cmd == ":spawn abrams" then
    task.spawn(spawnTank, player)
  elseif cmd == ":cleanup vehicles" then
    for _, entry in pairs(Tanks) do
      pcall(function() entry.car:Destroy() end)
    end
    Tanks = {}
    pcall(announce, "All Abrams removed.", player)
  end
end)
