-- ===== CFrame helpers =====
local function rx(deg) return CFrame.Angles(math.rad(deg), 0, 0) end
local function ry(deg) return CFrame.Angles(0, math.rad(deg), 0) end
local function rz(deg) return CFrame.Angles(0, 0, math.rad(deg)) end

-- ===== Weld helpers =====
local function weldConstraint(part0, part1)
  local w = Instance.new("WeldConstraint")
  w.Part0 = part0
  w.Part1 = part1
  w.Parent = part0
  return w
end

local function weld(part0, part1, c0)
  local w = Instance.new("Weld")
  w.Part0 = part0
  w.Part1 = part1
  w.C0 = c0 or CFrame.new()
  w.C1 = CFrame.new()
  w.Parent = part0
  return w
end

-- ===== Build kit =====
local function newKit(model, root)
  local k = { parts = {}, wheels = {} }
  function k.at(x, y, z) return root * CFrame.new(x*SCALE, y*SCALE, z*SCALE) end
  function k.off(x, y, z) return CFrame.new(x*SCALE, y*SCALE, z*SCALE) end
  function k.make(class, name, size, cf, color, material, weldTo, shape)
    local p = Instance.new(class)
    p.Name = name
    if shape then p.Shape = shape end
    p.Size = size * SCALE
    p.CFrame = cf
    p.Color = color
    p.Material = material
    p.Anchored = true
    p.CanCollide = false
    p.CanTouch = false
    p.Massless = true
    p.Parent = model
    f(p)
    if weldTo then weldConstraint(weldTo, p) end
    table.insert(k.parts, p)
    return p
  end
  function k.block(name, sx, sy, sz, cf, color, material, weldTo)
    return k.make("Part", name, Vector3.new(sx, sy, sz), cf, color, material, weldTo, false)
  end
  function k.wedge(name, sx, sy, sz, cf, color, material, weldTo)
    return k.make("WedgePart", name, Vector3.new(sx, sy, sz), cf, color, material, weldTo, false)
  end
  function k.cylinder(name, length, diameter, cf, color, material, weldTo)
    return k.make("Part", name, Vector3.new(length, diameter, diameter), cf, color, material, weldTo, CYL)
  end
  function k.glass(name, sx, sy, sz, cf, weldTo)
    local p = k.make("Part", name, Vector3.new(sx, sy, sz), cf, COLOR.GLASS, MAT.GLASS, weldTo, false)
    p.Transparency = 0.4
    return p
  end
  function k.seat(class, name, cf, weldTo)
    local s = k.make(class, name, Vector3.new(1.5, 0.4, 1.4), cf, COLOR.SEAT, MAT.FABRIC, weldTo, false)
    s.CanCollide = true
    s.CanTouch = true
    return s
  end
  return k
end

-- ===== Seat prompt =====
local function addSeatPrompt(seat)
  local ok = pcall(function()
    local prompt = Instance.new("ProximityPrompt")
    prompt.Name = "SitPrompt"
    prompt.ActionText = "Sit"
    prompt.ObjectText = seat.Name
    prompt.KeyboardKeyCode = Enum.KeyCode.E
    prompt.HoldDuration = 0
    prompt.MaxActivationDistance = 8
    prompt.RequiresLineOfSight = false
    prompt.Parent = seat
    prompt.Triggered:Connect(function(who)
      if seat.Occupant then return end
      pcall(function()
        local char = who.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum then seat:Sit(hum) end
      end)
    end)
  end)
  return ok
end

-- ===== BUILD THE HUMVEE =====

      at(wo.x, DIM.WHEEL_R, wo.z), COLOR.RUBBER, MAT.RUBBER, false, CYL)
    wheel.CanCollide = true
    wheel.Massless = false
    wheel.CustomPhysicalProperties = iceFriction
    -- detail parts weld to the wheel and follow it (they use WeldConstraint)
    k.cylinder("Rim_" .. name, 0.2, 2.0, at(wo.x + s * 0.62, DIM.WHEEL_R, wo.z), COLOR.TAN, MAT.SMOOTH, wheel)
    k.cylinder("Hub_" .. name, 0.3, 0.85, at(wo.x + s * 0.72, DIM.WHEEL_R, wo.z), COLOR.TAN_DARK, MAT.METAL, wheel)
    for lug = 0, 7 do
      local a = lug * math.pi / 4
      k.cylinder("LugNut", 0.12, 0.13,
        at(wo.x + s * 0.78, DIM.WHEEL_R + 0.62 * math.cos(a), wo.z + 0.62 * math.sin(a)),
        COLOR.STEEL, MAT.METAL, wheel)
    end
    -- Animatable Weld from chassis to wheel.
    local baseC0 = off(wo.x, DIM.WHEEL_R - DIM.CHASSIS_Y, wo.z - DIM.CHASSIS_Z)
    local wheelWeld = weld(chassis, wheel, baseC0)
    table.insert(k.wheels, {
      weld = wheelWeld,
      base = baseC0,
      front = wo.front,
      radius = DIM.WHEEL_R * SCALE,   -- actual stud radius
    })
  end

  -- ===== SIDE BODY =====

  k.block("ShieldCenter", 0.7, 0.95, 0.15, at(0, DIM.RING_TOP + 0.475, shieldZ), COLOR.TAN, MAT.SMOOTH, turretPivot)
  k.block("ShieldBridge", 0.7, 0.35, 0.15, at(0, DIM.RING_TOP + 1.825, shieldZ), COLOR.TAN, MAT.SMOOTH, turretPivot)
  for _, s in ipairs({ -1, 1 }) do
    k.block("ShieldWing", 1.3, 1.8, 0.15, at(s * 1.96, DIM.RING_TOP + 0.9, shieldZ + 0.46) * ry(-45 * s), COLOR.TAN, MAT.SMOOTH, turretPivot)
  end
  k.cylinder("Pintle", 0.9, 0.28, at(0, DIM.RING_TOP + 0.45, DIM.TURRET_Z - 0.2) * rz(90), COLOR.STEEL, MAT.METAL, turretPivot)

  local gunnerSeat = k.seat("VehicleSeat", "GunnerSeat", at(0, DIM.GUNNER_Y, DIM.GUNNER_Z), turretPivot)
  gunnerSeat.MaxSpeed = 0
  gunnerSeat.Torque = 0
  gunnerSeat.TurnSpeed = 0
  gunnerSeat.HeadsUpDisplay = false
  table.insert(seats, gunnerSeat)

  -- ===== GUN DETAIL (welded to gunPivot) =====
  k.block("Cradle", 0.5, 0.2, 0.7, at(0, 7.45, 0.95), COLOR.STEEL, MAT.METAL, gunPivot)
  k.block("Receiver", 0.42, 0.5, 1.6, at(0, DIM.GUN_Y, 0.9), COLOR.GUN, MAT.METAL, gunPivot)
  k.block("FeedCover", 0.44, 0.1, 1.1, at(0, DIM.GUN_Y + 0.3, 0.7), COLOR.STEEL, MAT.METAL, gunPivot)
  k.cylinder("BarrelJacket", 0.7, 0.32, at(0, DIM.GUN_Y, -0.25) * ry(90), COLOR.GUN, MAT.METAL, gunPivot)
  k.cylinder("Barrel", 3.0, 0.17, at(0, DIM.GUN_Y, -2.1) * ry(90), COLOR.GUN, MAT.METAL, gunPivot)
  k.cylinder("FlashHider", 0.3, 0.24, at(0, DIM.GUN_Y, -3.6) * ry(90), COLOR.GUN, MAT.METAL, gunPivot)
  k.block("GripPlate", 0.42, 0.34, 0.08, at(0, DIM.GUN_Y, 1.74), COLOR.GUN, MAT.METAL, gunPivot)
  for _, s in ipairs({ -1, 1 }) do
    k.cylinder("SpadeGrip", 0.35, 0.12, at(s * 0.14, DIM.GUN_Y - 0.05, 1.9) * rz(90), COLOR.GUN, MAT.METAL, gunPivot)
  end
  k.block("AmmoCan", 0.45, 0.6, 0.95, at(-0.5, DIM.GUN_Y - 0.1, 0.85), COLOR.OLIVE, MAT.METAL, gunPivot)

  -- ===== UNANCHOR EVERYTHING =====
  for _, p in ipairs(k.parts) do
    p.Anchored = false
  end
  turretPivot.Anchored = false
  gunPivot.Anchored = false

  -- Seat prompts. (fix) Uses this Humvee's own seat variables, never a search by name, so a seat
  -- called "DriverSeat" in another vehicle can never be picked up by mistake.
  for _, s in ipairs(seats) do addSeatPrompt(s) end

  -- (fix) Roblox puts a vehicle that sits still "to sleep", and a sleeping vehicle can ignore the
  -- velocity the drive loop sets, so it feels locked after you get out and back in. When someone gets
  -- in, it is lifted 0.05 studs, which wakes it. Only a CFrame nudge, no push, at most once every 2 s.
  local lastWake = 0
  local function wake()
    if tick() - lastWake < 2 then return end
    lastWake = tick()
    pcall(function() chassis.CFrame = chassis.CFrame + Vector3.new(0, 0.05, 0) end)
  end

  -- ===== DRIVE LOOP (simple) =====
  task.spawn(function()
    local speed = 0
    local steerAngle = 0        -- current front-wheel steer in radians
    local spinAccum = 0         -- accumulated wheel roll in radians
    local shownSpin, shownSteer = 0, 0
    local last = tick()
    local wasManned = false

    while car.Parent do
      local driving = driverSeat.Occupant ~= nil
      -- (fix) full speed only while driven or still rolling; four checks a second when parked
      if driving or speed ~= 0 or steerAngle ~= 0 then task.wait(0.03) else task.wait(0.25) end
      -- (fix) time step from tick(): if task.wait returns nothing in this game, "speed * dt" crashed
      -- the loop on its first tick (the same bug the Stryker had)
      local now = tick()
      local dt = math.min(now - last, 0.3)
      last = now

      local manned = driving or gunnerSeat.Occupant ~= nil
      if manned and not wasManned then wake() end
      wasManned = manned

      -- Read input
      if driving then
        local throttle = driverSeat.Throttle
        local steer    = driverSeat.Steer

        local targetSpeed = throttle * MAX_SPEED
        if throttle < 0 then targetSpeed = throttle * REVERSE_SPEED end
        speed = speed + (targetSpeed - speed) * ACCELERATION
        if throttle == 0 and math.abs(speed) < 0.05 then speed = 0 end

        local targetSteer = -steer * math.rad(MAX_STEER_DEG)
        steerAngle = steerAngle + (targetSteer - steerAngle) * STEER_RATE
        if steer == 0 and math.abs(steerAngle) < 0.005 then steerAngle = 0 end
      else
        speed = speed * (1 - BRAKE_DECEL)
        if math.abs(speed) < 0.05 then speed = 0 end
        steerAngle = steerAngle * 0.7
        if math.abs(steerAngle) < 0.005 then steerAngle = 0 end
      end

      -- Apply to chassis
      pcall(function()
        chassis.AssemblyLinearVelocity =
          chassis.CFrame.LookVector * speed + Vector3.new(0, chassis.AssemblyLinearVelocity.Y, 0)
        local speedRatio = math.clamp(math.abs(speed) / MAX_SPEED, 0.05, 1)
        local steerInput = 0
        if driving then steerInput = driverSeat.Steer end
        chassis.AssemblyAngularVelocity = Vector3.new(0, -steerInput * TURN_RATE * speedRatio, 0)
      end)

      -- Wheel roll: distance traveled divided by radius gives radians.
      spinAccum = spinAccum - (speed * dt) / (DIM.WHEEL_R * SCALE)

      -- Update each wheel (front wheels steer, all wheels spin), only when something changed
      if math.abs(spinAccum - shownSpin) > 0.02 or steerAngle ~= shownSteer then
        shownSpin, shownSteer = spinAccum, steerAngle
        pcall(function()
          for _, w in ipairs(k.wheels) do
            if w.front then
              w.weld.C0 = w.base * CFrame.Angles(0, steerAngle, 0) * CFrame.Angles(spinAccum, 0, 0)
            else
              w.weld.C0 = w.base * CFrame.Angles(spinAccum, 0, 0)
            end
          end
        end)
      end
    end
  end)

  -- ===== TURRET LOOP (simple) =====
  task.spawn(function()
    local yaw = 0
    local pitch = 0
    local last = tick()
    while car.Parent do
      -- (fix) four checks a second while nobody is on the gun
      if gunnerSeat.Occupant then task.wait(0.03) else task.wait(0.25) end
      local now = tick()
      local dt = math.min(now - last, 0.1) -- (fix) real time step, so aiming speed doesn't change with lag
      last = now
      if gunnerSeat.Occupant then
        local throttle = gunnerSeat.Throttle
        local steer    = gunnerSeat.Steer
        if steer ~= 0 or throttle ~= 0 then
          yaw   = (yaw - steer * TRAVERSE_SPEED * dt + math.pi) % (2 * math.pi) - math.pi
          pitch = math.clamp(pitch + throttle * ELEVATE_SPEED * dt, MIN_PITCH, MAX_PITCH)
          pcall(function()
            turretWeld.C0 = turretBase * CFrame.Angles(0, yaw, 0)
            elevationWeld.C0 = elevationBase * CFrame.Angles(pitch, 0, 0)
          end)
        end
      end
    end
  end)

  entry.car = car
  entry.chassis = chassis
  entry.driverSeat = driverSeat
  entry.gunnerSeat = gunnerSeat
  return true
end

-- ===== SPAWN / DESTROY =====
local function destroyHumvee(player)
  if Humvees[player] then
    pcall(function() Humvees[player].car:Destroy() end)
    Humvees[player] = nil
  end
end

local function spawnHumvee(player)
  destroyHumvee(player)
  local cf = getPlayerPosition(player)
  if not cf then return end
  if typeof(cf) == "Vector3" then cf = CFrame.new(cf) end

  local feetY = cf.Position.Y - 3
  local forward = Vector3.new(cf.LookVector.X, 0, cf.LookVector.Z)
  if forward.Magnitude < 0.1 then forward = Vector3.new(0, 0, -1) end
  forward = forward.Unit

  local spawnPos = Vector3.new(cf.Position.X, feetY, cf.Position.Z) + forward * 22
  local root = CFrame.new(spawnPos + Vector3.new(0, 2, 0), spawnPos + Vector3.new(0, 2, 0) + forward)

  local entry = {}
  Humvees[player] = entry

  local ok, err = pcall(buildHumvee, player, root, entry)
  if not ok then
    print("[Humvee] build failed: " .. tostring(err))
    pcall(function() entry.car:Destroy() end)
    Humvees[player] = nil
    return
  end

  local drop = root * CFrame.new(-1.6 * SCALE, DIM.SEAT_Y * SCALE + 3, -0.35 * SCALE)
  pcall(setPlayerPosition, player, drop)
  pcall(announce, player .. " deployed a Humvee. Driver: W/S + A/D. Press E on a seat to sit; the turret seat runs the gun.", player)
end

-- ===== COMMANDS =====
event("chatted", function(data)
  local player = data.Value[1]
  local message = data.Value[2]
  if not player or not message then return end
  local cmd = string.match(string.lower(message), "^%s*(.-)%s*$")

  if cmd == ":spawn humvee" then
    task.spawn(spawnHumvee, player)
  elseif cmd == ":cleanup vehicles" then
    for _, entry in pairs(Humvees) do
      pcall(function() entry.car:Destroy() end)
    end
    Humvees = {}
    pcall(announce, "All Humvees removed.", player)
  end
end)
