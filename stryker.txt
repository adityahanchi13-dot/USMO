-- ===== HELPERS =====
local function rx(d) return CFrame.Angles(math.rad(d), 0, 0) end
local function ry(d) return CFrame.Angles(0, math.rad(d), 0) end
local function rz(d) return CFrame.Angles(0, 0, math.rad(d)) end
local function weldConstraint(a, b)
  local w = Instance.new("WeldConstraint")
  w.Part0 = a; w.Part1 = b; w.Parent = a
  return w
end
local function weld(a, b, c0)
  local w = Instance.new("Weld")
  w.Part0 = a; w.Part1 = b; w.C0 = c0 or CFrame.new(); w.C1 = CFrame.new(); w.Parent = a
  return w
end
local function tell(msg, player) pcall(announce, msg, player) end

-- ===== BUILD KIT =====
local function newKit(car, root)
  local k = { parts = {}, wheels = {} }
  function k.at(x, y, z) return root * CFrame.new(x*SCALE, y*SCALE, z*SCALE) end
  function k.off(x, y, z) return CFrame.new(x*SCALE, y*SCALE, z*SCALE) end
  function k.facing(x, y, z, dx, dy, dz)
    local p = Vector3.new(x, y, z) * SCALE
    return root * CFrame.new(p, p + Vector3.new(dx, dy, dz))
  end
  function k.make(class, name, size, cf, color, mat, weldTo, shape)
    local p = Instance.new(class)
    p.Name = name
    if shape then p.Shape = shape end
    p.Size = size * SCALE
    p.CFrame = cf
    p.Color = color
    p.Material = mat
    p.Anchored = true
    p.CanCollide = false
    p.CanTouch = false
    p.Massless = true
    p.Parent = car
    f(p)
    if weldTo then weldConstraint(weldTo, p) end
    table.insert(k.parts, p)
    return p
  end
  function k.block(name, sx, sy, sz, cf, color, mat, weldTo)
    return k.make("Part", name, Vector3.new(sx, sy, sz), cf, color, mat, weldTo, false)
  end
  function k.wedge(name, sx, sy, sz, cf, color, mat, weldTo)
    return k.make("WedgePart", name, Vector3.new(sx, sy, sz), cf, color, mat, weldTo, false)
  end
  function k.cylinder(name, len, dia, cf, color, mat, weldTo)
    return k.make("Part", name, Vector3.new(len, dia, dia), cf, color, mat, weldTo, SHAPE.CYLINDER)
  end
  function k.bolt(cf, weldTo)
    return k.cylinder("Bolt", 0.06, 0.16, cf, COLOR.TAN_DARK, MAT.METAL, weldTo)
  end
  function k.seat(class, name, cf, weldTo)
    local s = k.make(class, name, Vector3.new(1.5, 0.4, 1.4), cf, COLOR.SEAT, MAT.FABRIC, weldTo, false)
    s.CanCollide = true
    s.CanTouch = true
    return s
  end
  function k.anchor(name, x, y, z, weldTo)
    local p = k.block(name, 0.4, 0.4, 0.4, k.at(x, y, z), COLOR.BLACK, MAT.SMOOTH, weldTo)
    p.Transparency = 1
    return p
  end
  function k.joint(name, p0, p1, offset, c1)
    local j = Instance.new("Weld")
    j.Name = name; j.Part0 = p0; j.Part1 = p1
    j.C0 = offset; j.C1 = c1 or CFrame.new(); j.Parent = p0
    return j
  end
  return k
end

-- ===== WHEEL =====
local function buildWheel(k, name, x, z, steer)
  local s = 1
  if x < 0 then s = -1 end
  local r, w, y = DIM.WHEEL_R, DIM.WHEEL_W, DIM.WHEEL_R
  local tyre = k.cylinder(name, w, r * 2, k.at(x, y, z), COLOR.RUBBER, MAT.RUBBER, false)
  tyre.CanCollide = true
  tyre.CustomPhysicalProperties = PhysicalProperties.new(0, 0, 0, 0, 0)
  local face = x + s * w / 2
  k.cylinder("Rim", 0.1, r * 1.2, k.at(face, y, z), COLOR.TAN, MAT.SMOOTH, tyre)
  k.cylinder("Hub", 0.28, r * 0.55, k.at(face + s * 0.12, y, z), COLOR.TAN_DARK, MAT.METAL, tyre)
  k.cylinder("HubCap", 0.12, r * 0.25, k.at(face + s * 0.3, y, z), COLOR.STEEL, MAT.METAL, tyre)
  k.cylinder("TyreValve", 0.2, 0.12, k.at(face + s * 0.25, y + r * 0.25, z + r * 0.12), COLOR.BLACK, MAT.METAL, tyre)
  for lug = 0, 7 do
    local a = lug * math.pi / 4
    k.cylinder("LugNut", 0.12, 0.15,
      k.at(face + s * 0.1, y + r * 0.42 * math.cos(a), z + r * 0.42 * math.sin(a)),
      COLOR.STEEL, MAT.METAL, tyre)
  end
  local j = k.joint("WheelJoint", k.chassis, tyre, k.off(x, y - DIM.CHASSIS_Y, z - DIM.CHASSIS_Z), CFrame.new())
  table.insert(k.wheels, { weld = j, base = j.C0, steer = steer, radius = DIM.WHEEL_R * SCALE })
end

-- ===== HULL =====

  local at, block, wedge, cylinder = k.at, k.block, k.wedge, k.cylinder
  local hx, lx = DIM.HULL_X, DIM.LOWER_X
  local sp, roof = DIM.SPONSON_Y, DIM.ROOF_Y
  local bz, tail = DIM.BULKHEAD_Z, DIM.TAIL
  local troopLen = tail - bz
  local troopZ = (tail + bz) / 2
  local frontLen = DIM.GLACIS_Z - DIM.NOSE
  local frontZ = (DIM.GLACIS_Z + DIM.NOSE) / 2
  block("Floor", lx * 2, DIM.FLOOR_Y - DIM.BELLY_Y, tail + 9.4,
    at(0, (DIM.FLOOR_Y + DIM.BELLY_Y) / 2, (tail - 9.4) / 2), COLOR.TAN_DARK, MAT.SMOOTH, c)
  block("LowerFront", lx * 2, sp - DIM.FLOOR_Y, bz + 9.4,
    at(0, (sp + DIM.FLOOR_Y) / 2, (bz - 9.4) / 2), COLOR.TAN, MAT.SMOOTH, c)
  wedge("LowerGlacis", lx * 2, sp - DIM.BELLY_Y, 2.2,
    at(0, (sp + DIM.BELLY_Y) / 2, -10.5) * rz(180), COLOR.TAN, MAT.SMOOTH, c)
  block("Bow", hx * 2, DIM.NOSE_Y - sp, frontLen, at(0, (DIM.NOSE_Y + sp) / 2, frontZ), COLOR.TAN, MAT.SMOOTH, c)
  wedge("UpperGlacis", hx * 2, roof - DIM.NOSE_Y, frontLen,
    at(0, (roof + DIM.NOSE_Y) / 2, frontZ), COLOR.TAN, MAT.SMOOTH, c)
  block("FrontSection", hx * 2, roof - sp, bz - DIM.GLACIS_Z,
    at(0, (roof + sp) / 2, (bz + DIM.GLACIS_Z) / 2), COLOR.TAN, MAT.SMOOTH, c)
  for _, s in ipairs({ -1, 1 }) do
    local wall = block("LowerWall", 0.3, sp - DIM.FLOOR_Y, troopLen,
      at(s * (lx - 0.15), (sp + DIM.FLOOR_Y) / 2, troopZ), COLOR.TAN, MAT.SMOOTH, c)
    wall.CanCollide = true
    block("SponsonFloor", hx - lx, 0.2, troopLen, at(s * (hx + lx) / 2, sp + 0.1, troopZ), COLOR.TAN, MAT.SMOOTH, c)
    block("UpperWall", 0.3, roof - sp, troopLen,
      at(s * (hx - 0.15), (roof + sp) / 2, troopZ), COLOR.TAN, MAT.SMOOTH, c)
    block("WallLiner", 0.05, sp - DIM.FLOOR_Y - 0.1, troopLen - 0.4,
      at(s * (lx - 0.33), (sp + DIM.FLOOR_Y) / 2, troopZ), COLOR.INTERIOR, MAT.SMOOTH, c)
    block("WallLiner", 0.05, roof - sp - 0.5, troopLen - 0.4,
      at(s * (hx - 0.33), (roof + sp) / 2 - 0.1, troopZ), COLOR.INTERIOR, MAT.SMOOTH, c)
    block("RearPillar", hx - DIM.RAMP_W / 2, roof - sp, 0.3,
      at(s * (hx + DIM.RAMP_W / 2) / 2, (roof + sp) / 2, tail - 0.15), COLOR.TAN, MAT.SMOOTH, c)
    local post = block("RearPillar", lx - DIM.RAMP_W / 2, sp - DIM.FLOOR_Y, 0.3,
      at(s * (lx + DIM.RAMP_W / 2) / 2, (sp + DIM.FLOOR_Y) / 2, tail - 0.15), COLOR.TAN, MAT.SMOOTH, c)
    post.CanCollide = true

  block("RoofLiner", hx * 2 - 0.8, 0.05, troopLen - 0.4, at(0, roof - 0.33, troopZ), COLOR.INTERIOR, MAT.SMOOTH, c)

    at(0, (roof + DIM.FLOOR_Y) / 2 - 0.1, bz + 0.03), COLOR.INTERIOR, MAT.SMOOTH, c)
  bulkhead.CanCollide = true
  for _, lz in ipairs({ 2.5, 6.5, 10.0 }) do
    block("DomeLight", 0.6, 0.08, 0.3, at(0, roof - 0.38, lz), COLOR.LAMP, MAT.NEON, c)

    cylinder("Axle", DIM.WHEEL_X * 2 - DIM.WHEEL_W, 0.45, at(0, DIM.WHEEL_R, z), COLOR.STEEL, MAT.METAL, c)
    block("Differential", 1.2, 0.9, 1.0, at(0, DIM.WHEEL_R, z), COLOR.STEEL, MAT.METAL, c)
  end

-- ===== ARMOUR =====
local function buildArmour(k)
  local c = k.chassis
  local at, block, bolt = k.at, k.block, k.bolt
  local hx = DIM.HULL_X
  local rise = DIM.ROOF_Y - DIM.NOSE_Y
  local run = DIM.GLACIS_Z - DIM.NOSE
  local slope = math.deg(math.atan(rise / run))
  for _, s in ipairs({ -1, 1 }) do
    for col = 0, 6 do
      local z = -6.95 + col * 2.62 + 1.275
      for row = 0, 1 do
        local y = 3.95 + row * 1.37 + 0.65
        if not (s > 0 and row == 0 and col <= 1) then
          block("ArmourTile", 0.12, 1.3, 2.55, at(s * (hx + 0.06), y, z), COLOR.TAN, MAT.SMOOTH, c)
          for _, dy in ipairs({ -0.45, 0.45 }) do
            for _, dz in ipairs({ -1.0, 1.0 }) do
              bolt(at(s * (hx + 0.15), y + dy, z + dz), c)
            end
          end
        end
      end
    end
    block("BowTile", 0.12, 1.4, 3.6, at(s * (hx + 0.06), 4.5, -9.6), COLOR.TAN, MAT.SMOOTH, c)
    for _, dz in ipairs({ -1.4, 0, 1.4 }) do
      bolt(at(s * (hx + 0.15), 4.95, -9.6 + dz), c)
      bolt(at(s * (hx + 0.15), 4.05, -9.6 + dz), c)
    end
  end
  for ix = -1, 1 do
    for iz = 0, 1 do
      local z = DIM.NOSE + 1.2 + iz * 2.1
      local y = DIM.NOSE_Y + rise * (z - DIM.NOSE) / run + 0.06
      local frame = at(ix * 2.75, y, z) * rx(-slope)
      block("GlacisTile", 2.6, 0.12, 2.0, frame, COLOR.TAN, MAT.SMOOTH, c)
      for _, dx in ipairs({ -1.05, 1.05 }) do
        for _, dz in ipairs({ -0.8, 0.8 }) do
          bolt(frame * k.off(dx, 0.08, dz) * rz(90), c)
        end
      end
    end
  end
end

-- ===== BUILD =====
local function buildStryker(owner, root, entry)
  local car = Instance.new("Model")
  car.Name = owner .. "_Stryker"
  f(car)

  local k = newKit(car, root)

  local chassis = k.block("Chassis", 3.0, 0.4, 4.0, k.at(0, DIM.CHASSIS_Y, DIM.CHASSIS_Z), COLOR.BLACK, MAT.SMOOTH, false)
  chassis.Massless = false
  -- (DeepSeek fix) no collider on the chassis: only the wheels touch the ground
  chassis.CustomPhysicalProperties = PhysicalProperties.new(DIM.ROOT_DENSITY, 0.3, 0, 1, 1)
  chassis.Transparency = 1
  k.chassis = chassis

  -- Build every part BEFORE the unanchor loop, including the ramp toggle anchor.
  buildHull(k)
  buildArmour(k)
  buildBow(k)
  buildRoof(k)
  buildSides(k)
  buildRear(k)
  buildInterior(k)
  for i, z in ipairs(DIM.AXLES) do
    local steer = 0
    if i == 1 then steer = 1
    elseif i == 2 then steer = 0.55 end
    local name = "RoadWheel"
    if i >= 3 then name = "RoadWheelRear" end
    buildWheel(k, name, -DIM.WHEEL_X, z, steer)
    buildWheel(k, name, DIM.WHEEL_X, z, steer)
  end
  buildRamp(k)
  buildRws(k)
  buildSeats(k)

  -- The ramp toggle anchor part — created BEFORE the unanchor loop.
  local rampAnchor = k.block("RampToggle", 0.4, 0.4, 0.4,
    k.at(0, 4.0, DIM.TAIL + 1.6), COLOR.BLACK, MAT.SMOOTH, chassis)
  rampAnchor.Transparency = 1

  -- Now unanchor everything.
  for _, p in ipairs(k.parts) do
    p.Anchored = false
  end
  chassis.Anchored = false
  rampAnchor.Anchored = false

  entry.owner = owner
  entry.car = car
  entry.chassis = chassis
  entry.wheels = k.wheels
  entry.ramp = k.ramp
  entry.rws = k.rws
  entry.driverSeat = k.driverSeat
  entry.commanderSeat = k.commanderSeat
  entry.troopSeats = k.troopSeats
  entry.rampOpen = false
  entry.rampShown = 0
  entry.yaw = 0
  entry.pitch = 0

  -- ===== SEAT PROMPTS =====
  local function addSeatPrompt(seat)
    pcall(function()
      local prompt = Instance.new("ProximityPrompt")
      prompt.Name = "SitPrompt"
      prompt.ActionText = "Sit"
      prompt.ObjectText = seat.Name
      prompt.KeyboardKeyCode = Enum.KeyCode.E
      prompt.HoldDuration = 0
      prompt.MaxActivationDistance = 8
      prompt.RequiresLineOfSight = false
      prompt.Parent = seat
      prompt.Triggered:Connect(function(who)
        if seat.Occupant then return end
        local char = who.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum then seat:Sit(hum) end
      end)
    end)
  end
  addSeatPrompt(k.driverSeat)
  addSeatPrompt(k.commanderSeat)
  for _, s in ipairs(k.troopSeats) do addSeatPrompt(s) end

  -- ===== RAMP PROMPT =====
  pcall(function()
    local prompt = Instance.new("ProximityPrompt")
    prompt.Name = "RampPrompt"
    prompt.ActionText = "Lower ramp"
    prompt.ObjectText = "Rear ramp"
    prompt.KeyboardKeyCode = Enum.KeyCode.F
    prompt.HoldDuration = 0
    prompt.MaxActivationDistance = 8
    prompt.RequiresLineOfSight = false
    prompt.Parent = rampAnchor
    prompt.Triggered:Connect(function()
      entry.rampOpen = not entry.rampOpen
      prompt.ActionText = entry.rampOpen and "Raise ramp" or "Lower ramp"
    end)
  end)

  -- ===== DRIVE LOOP =====
  -- Kept light on purpose: the game can stop an addon that does too much work, which freezes the
  -- vehicle and stops every command (respawn included). Full speed (every 0.03 s) only while someone
  -- is driving or it is still rolling; four checks a second when parked. Wheels redraw 10 times a second.
  local sentinels = { chassis, k.parts[2], k.parts[math.floor(#k.parts / 2)], k.parts[#k.parts] }
  task.spawn(function()
    local speed = 0
    local steerAngle = 0
    local rolled = 0
    local shownRolled, shownSteer = 0, 0
    local last = tick()
    local lastOwner = 0
    local lastPrint = 0
    local lastWheels = 0
    local toldAnchor = false
    local spawnedAt = tick()
    local wasManned = false
    local stuckFor = 0
    local toldWake = false
    -- Roblox puts a vehicle that sits still "to sleep", and a sleeping vehicle can ignore the velocity
    -- this loop sets: it looks locked after you get out and back in. A tiny nudge wakes it up.
    local function wake(why)
      pcall(function() chassis.CFrame = chassis.CFrame + Vector3.new(0, 0.05, 0) end)
      pcall(function() chassis:ApplyImpulse(Vector3.new(0, chassis.AssemblyMass * 0.5, 0)) end)
      if not toldWake then
        toldWake = true
        print("[Stryker] woke the hull (" .. why .. ")")
      end
    end
    while car.Parent do
      local driving = k.driverSeat.Occupant ~= nil
      -- someone just got in (driver or commander): wake it before they try to move or aim
      local manned = driving or k.commanderSeat.Occupant ~= nil
      if manned and not wasManned then wake("someone got in") end
      wasManned = manned
      if driving or speed ~= 0 then task.wait(0.03) else task.wait(0.25) end
      local now = tick()
      local dt = math.min(now - last, 0.3)
      last = now

      -- if something anchored the vehicle, release every part and say what happened (once)
      local anchored = false
      for _, p in ipairs(sentinels) do
        if p.Anchored then anchored = true end
      end
      if anchored then
        local count = 0
        for _, p in ipairs(k.parts) do
          if p.Anchored then
            p.Anchored = false
            count = count + 1
          end
        end
        if not toldAnchor then
          toldAnchor = true
          print("[Stryker] something anchored " .. count .. " parts, " .. math.floor(now - spawnedAt)
            .. " s after spawn (model name now: " .. tostring(car.Name) .. "). Released them.")
        end
      end

      -- keep the physics on the server, so the velocity set below actually takes effect
      if now - lastOwner >= 0.5 then
        lastOwner = now
        pcall(function()
          if chassis:GetNetworkOwner() ~= nil then chassis:SetNetworkOwner(nil) end
        end)
      end

      local throttle, steer = 0, 0
      if driving then
        throttle = k.driverSeat.Throttle
        steer    = k.driverSeat.Steer
      end
      local target = throttle * MAX_SPEED
      if throttle < 0 then target = throttle * REVERSE_SPEED end
      local accel = ACCELERATION
      if throttle == 0 then accel = BRAKE_DECEL end
      speed = speed + (target - speed) * accel
      if throttle == 0 and math.abs(speed) < 0.05 then speed = 0 end

      steerAngle = steerAngle + (-steer * MAX_STEER - steerAngle) * STEER_RATE
      if steer == 0 and math.abs(steerAngle) < 0.005 then steerAngle = 0 end

      pcall(function()
        chassis.AssemblyLinearVelocity = chassis.CFrame.LookVector * speed
          + Vector3.new(0, chassis.AssemblyLinearVelocity.Y, 0)
        local speedRatio = math.clamp(math.abs(speed) / MAX_SPEED, 0.05, 1)
        chassis.AssemblyAngularVelocity = Vector3.new(0, -steer * TURN_RATE * speedRatio, 0)
      end)

      -- still stuck while the driver is asking it to move: wake it again
      if driving and math.abs(speed) > 5 then
        local real = 0
        pcall(function()
          local v = chassis.AssemblyLinearVelocity
          real = Vector3.new(v.X, 0, v.Z).Magnitude
        end)
        if real < 1 then stuckFor = stuckFor + dt else stuckFor = 0 end
        if stuckFor > 0.3 then
          stuckFor = 0
          wake("not moving while the driver holds W/S")
        end
      else
        stuckFor = 0
      end

      -- (debug) every 5 s while someone drives: seconds since spawn, anchored?, wanted speed, real speed
      if driving and now - lastPrint >= 5 then
        lastPrint = now
        pcall(function()
          print("[Stryker] alive " .. math.floor(now - spawnedAt) .. " s", chassis.Anchored, math.floor(speed),
            math.floor(chassis.AssemblyLinearVelocity.Magnitude))
        end)
      end

      if math.abs(speed) > 3 then entry.rampOpen = false end

      rolled = rolled + speed * dt
      if now - lastWheels >= 0.1 and (math.abs(rolled - shownRolled) > 0.05 or steerAngle ~= shownSteer) then
        lastWheels = now
        shownRolled, shownSteer = rolled, steerAngle
        pcall(function()
          for _, w in ipairs(k.wheels) do
            w.weld.C0 = w.base * CFrame.Angles(0, steerAngle * w.steer, 0) * CFrame.Angles(-rolled / w.radius, 0, 0)
          end
        end)
      end
    end
  end)

  -- ===== RAMP + TURRET LOOP =====
  task.spawn(function()
    local yaw = 0
    local pitch = 0
    local ramp = 0
    local last = tick()
    while car.Parent do
      local now = tick()
      local dt = math.min(now - last, 0.2)
      last = now
      -- full speed only while the ramp moves or someone is on the gun; otherwise four checks a second
      if k.commanderSeat.Occupant or math.abs(ramp - (entry.rampOpen and 1 or 0)) > 0.001 then
        task.wait(0.03)
      else
        task.wait(0.25)
      end

      -- Ramp animation
      local rampGoal = entry.rampOpen and 1 or 0
      if math.abs(ramp - rampGoal) > 0.001 then
        local step = dt / RAMP_TIME
        if ramp < rampGoal then ramp = math.min(rampGoal, ramp + step)
        else ramp = math.max(rampGoal, ramp - step) end
        pcall(function()
          entry.ramp.weld.C0 = entry.ramp.base * CFrame.Angles(RAMP_OPEN_ANGLE * ramp, 0, 0)
        end)
      end
      entry.rampShown = ramp

      -- Turret aim
      if k.commanderSeat.Occupant then
        local throttle = k.commanderSeat.Throttle
        local steer    = k.commanderSeat.Steer
        if steer ~= 0 or throttle ~= 0 then
          yaw = yaw - steer * TRAVERSE_SPEED * dt
          yaw = (yaw + math.pi) % (2 * math.pi) - math.pi
          pitch = math.clamp(pitch + throttle * ELEVATE_SPEED * dt, MIN_PITCH, MAX_PITCH)
          entry.yaw = yaw
          entry.pitch = pitch
          pcall(function()
            entry.rws.yaw.C0   = entry.rws.yawBase   * CFrame.Angles(0, yaw, 0)
            entry.rws.pitch.C0 = entry.rws.pitchBase * CFrame.Angles(pitch, 0, 0)
          end)
        end
      end
    end
  end)

  return true
end

-- ===== SPAWN / DESTROY =====
local function destroyStryker(player)
  if Strykers[player] then
    pcall(function() Strykers[player].car:Destroy() end)
    Strykers[player] = nil
  end
end

local function spawnStryker(player)
  destroyStryker(player)
  local cf = getPlayerPosition(player)
  if not cf then return end
  if typeof(cf) == "Vector3" then cf = CFrame.new(cf) end

  local feetY = cf.Position.Y - 3
  local forward = Vector3.new(cf.LookVector.X, 0, cf.LookVector.Z)
  if forward.Magnitude < 0.1 then forward = Vector3.new(0, 0, -1) end
  forward = forward.Unit

  local spawnPos = Vector3.new(cf.Position.X, feetY, cf.Position.Z) + forward * 26
  local root = CFrame.new(spawnPos + Vector3.new(0, 2, 0), spawnPos + Vector3.new(0, 2, 0) + forward)

  local entry = {}
  Strykers[player] = entry

  local ok, err = pcall(buildStryker, player, root, entry)
  if not ok then
    print("[Stryker] build failed: " .. tostring(err))
    Strykers[player] = nil
    return
  end

  local drop = root * CFrame.new(DIM.DRIVER_X * SCALE, DIM.SEAT_Y * SCALE + 3, DIM.DRIVER_Z * SCALE)
  pcall(setPlayerPosition, player, drop)
  pcall(announce, player .. " deployed a Stryker. Driver: W/S + A/D. F at the rear toggles the ramp. Commander seat runs the .50.", player)
end

-- ===== COMMANDS =====
event("chatted", function(data)
  local player = data.Value[1]
  local message = data.Value[2]
  if not player or not message then return end
  local cmd = string.lower(message)

  if cmd == ":spawn stryker" then
    task.spawn(spawnStryker, player)
  elseif cmd == ":stryker alive" then
    print("[Stryker] addon is running. " .. tostring(player) .. " asked.")
    pcall(announce, "Stryker addon is running.", player)
  elseif cmd == ":cleanup vehicles" then
    for _, entry in pairs(Strykers) do
      pcall(function() entry.car:Destroy() end)
    end
    Strykers = {}
    pcall(announce, "All Strykers removed.", player)
  end
end)

print("[Stryker] simple version loaded. :spawn stryker to build one.")
